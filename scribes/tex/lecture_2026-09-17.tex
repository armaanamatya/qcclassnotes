In this lecture, we review linear algebra: the product $Ax$ as a combination of the columns of $A$,
linear independence, column rank and row rank, and orthogonality.

\section{The Product \texorpdfstring{$Ax$}{Ax} as a Combination of Columns}

Let $A \in \mathbb{R}^{3 \times 3}$ be the matrix with columns
$v_1 = (2, 3, 5)^{T}$, $v_2 = (1, 1, 7)^{T}$, $v_3 = (3, 4, 12)^{T}$. Then
\begin{equation}\label{eq:columns-0917}
    A = \begin{bmatrix} 2 & 1 & 3 \\ 3 & 1 & 4 \\ 5 & 7 & 12 \end{bmatrix}, \qquad
    Ax = \begin{bmatrix} 2x_1 + x_2 + 3x_3 \\ 3x_1 + x_2 + 4x_3 \\ 5x_1 + 7x_2 + 12x_3 \end{bmatrix}
    = x_1 v_1 + x_2 v_2 + x_3 v_3.
\end{equation}
Here $v_3 = v_1 + v_2$, so $Ax = (x_1 + x_3)\, v_1 + (x_2 + x_3)\, v_2$.

In general, an $m \times n$ matrix $A$ with columns $a_1, \dots, a_n$ defines the linear map
$T \colon \mathbb{R}^n \to \mathbb{R}^m$, $T(x) = Ax = x_1 a_1 + \dots + x_n a_n$. Its \emph{image}
is the \emph{column space} of $A$:
\[\operatorname{Im}(A) = \set{Ax \given x \in \mathbb{R}^n} = \operatorname{span}(a_1, \dots, a_n).\]

\section{Linear Independence and Rank}

\begin{definition}[Linear Independence]
    Vectors $v_1, \dots, v_k$ are \textbf{linearly independent} if
    \[c_1 v_1 + c_2 v_2 + \dots + c_k v_k = 0 \quad\Longrightarrow\quad c_1 = c_2 = \dots = c_k = 0.\]
    Otherwise they are \textbf{linearly dependent}. If $v_1, \dots, v_k$ are linearly independent,
    we say that a vector $v$ is \emph{independent of} them if $v, v_1, \dots, v_k$ are linearly independent.
\end{definition}

\begin{definition}[Column Rank and Row Rank]
    The \textbf{column rank} $\operatorname{rank}_c(A)$ of a matrix $A$ is the maximum number of linearly
    independent columns of $A$. The \textbf{row rank} $\operatorname{rank}_r(A)$ is the maximum number of
    linearly independent rows of $A$.
\end{definition}

\begin{lemma}[Column Rank Is the Dimension of the Image]\label{lem:colbasis-0917}
    Let $A$ be an $m \times n$ matrix with $r = \operatorname{rank}_c(A) \geq 1$. Then any $r$ linearly
    independent columns of $A$ form a basis of $\operatorname{Im}(A)$. So
    \[\operatorname{rank}_c(A) = \dim(\operatorname{Im}(A)).\]
\end{lemma}

\begin{exercise}\label{ex:example-rank-0917}
    For the matrix $A$ of \cref{eq:columns-0917}, find $\operatorname{rank}_c(A)$,
    $\operatorname{Im}(A)$ and $\operatorname{rank}_r(A)$, and express every row of $A$ as a combination
    of $[1\ 0\ 1]$ and $[0\ 1\ 1]$.
\end{exercise}

\begin{wsolution}
    \emph{Columns.} $v_1, v_2$ are independent: $a v_1 + b v_2 = 0$ gives $2a + b = 0$ and
    $3a + b = 0$, so $a = 0$ and then $b = 0$. The three columns are dependent, since
    $v_1 + v_2 - v_3 = 0$. So $\operatorname{rank}_c(A) = 2$, and by \cref{lem:colbasis-0917},
    $\operatorname{Im}(A) = \operatorname{span}(v_1, v_2)$, of dimension $2$.

    \emph{Rows.} Let $u_1 = [1\ 0\ 1]$ and $u_2 = [0\ 1\ 1]$, so that $a u_1 + b u_2 = [a\ \ b\ \ a + b]$.
    Then
    \[[2\ 1\ 3] = 2 u_1 + 1 u_2, \qquad [3\ 1\ 4] = 3 u_1 + 1 u_2, \qquad [5\ 7\ 12] = 5 u_1 + 7 u_2.\]
    The three rows are dependent, since $16\,[2\ 1\ 3] - 9\,[3\ 1\ 4] - [5\ 7\ 12] = 0$, so
    $\operatorname{rank}_r(A) \leq 2$. The
    first two rows are independent: $a\,[2\ 1\ 3] + b\,[3\ 1\ 4] = 0$ gives $a + b = 0$ (second entry)
    and $2a + 3b = 0$ (first entry), so $a = b = 0$. So $\operatorname{rank}_r(A) = 2$.
\end{wsolution}

\begin{theorem}[Row Rank Equals Column Rank]\label{thm:rank-equal-0917}
    For every $m \times n$ matrix $A$, $\operatorname{rank}_r(A) = \operatorname{rank}_c(A)$. This common
    value is called the \textbf{rank} of $A$.
\end{theorem}

\section{Orthogonality}

The results of this section hold over $\mathbb{R}$ and over $\mathbb{C}$. The inner product is
$\langle a, b \rangle = a \cdot b = \sum_{i=1}^{n} a_i b_i$ for $a, b \in \mathbb{R}^n$, and
$\langle a, b \rangle = \sum_{i=1}^{n} \overline{a_i} \, b_i$ (linear in $b$) for $a, b \in \mathbb{C}^n$.
In both cases $\langle a, a \rangle = \sum_{i=1}^{n} \abs{a_i}^2 = \norm{a}^2$, which is $0$ only for
$a = 0$.

\begin{definition}[Orthogonality]
    Two vectors $a$ and $b$ are \textbf{orthogonal} if $\langle a, b \rangle = 0$.
\end{definition}

\needspace{12\baselineskip} % keep the exercise and the start of its solution on one page
\begin{exercise}\label{ex:orth-ind-0917}\label{thm:orth-ind-0917}
    Let $v_1, \dots, v_k$ be linearly independent vectors in $\mathbb{R}^n$ or $\mathbb{C}^n$, and let
    $v \neq 0$ be orthogonal to each of them. Show that $v, v_1, \dots, v_k$ are linearly independent,
    so $v \notin \operatorname{span}(v_1, \dots, v_k)$.
\end{exercise}

\begin{proof}
    Suppose that $c \, v + c_1 v_1 + \dots + c_k v_k = 0$. Take the inner product of $v$ with
    both sides (it is linear in the second argument) and use $\langle v, v_i \rangle = 0$:
    \[0 = c \, \langle v, v \rangle + \sum_{i=1}^{k} c_i \, \langle v, v_i \rangle = c \, \norm{v}^2.\]
    Since $v \neq 0$, $\norm{v}^2 > 0$, so $c = 0$. Then $c_1 v_1 + \dots + c_k v_k = 0$, and
    the independence of $v_1, \dots, v_k$ gives $c_1 = \dots = c_k = 0$. If $v = \sum_{i} \alpha_i v_i$,
    then $1 \cdot v - \sum_{i} \alpha_i v_i = 0$ would be a combination with a nonzero coefficient
    that gives zero, so $v \notin \operatorname{span}(v_1, \dots, v_k)$.
\end{proof}

\paragraph{Remark.} The assumption $v \neq 0$ is needed. The zero vector is orthogonal to every vector,
but no set containing it is independent ($1 \cdot 0 = 0$ has a nonzero coefficient).

In this lecture, we state and prove Markov's inequality and Chebyshev's inequality, and we state the
Chernoff bound. We then look at the query complexity of search for classical and quantum algorithms,
and we analyze the birthday paradox.

\section{Markov's Inequality}

Markov's inequality uses only the expectation $\expectation{X}$ of the random variable, which is
also called the \emph{first moment}.

\begin{theorem}[Markov's Inequality]\label{thm:markov}
    Let $X \geq 0$ be a random variable. Then, for any $a > 0$,
    \[\prob{X \geq a \expectation{X}} \leq \frac{1}{a}.\]
    Equivalently, for any $b > 0$,
    \[\prob{X \geq b} \leq \frac{\expectation{X}}{b}.\]
\end{theorem}

\begin{proof}
    Let $I = 1$ if $X \geq b$ and $I = 0$ otherwise. Then $I \leq \frac{X}{b}$: if $X \geq b$ then
    $I = 1 \leq \frac{X}{b}$, and if $X < b$ then $I = 0 \leq \frac{X}{b}$ because $X \geq 0$.
    Taking expectations on both sides,
    \[\prob{X \geq b} = \prob{I = 1} = \expectation{I} \leq \expectation{\frac{X}{b}} = \frac{\expectation{X}}{b}.\]
    Taking $b = a \expectation{X}$ gives $\prob{X \geq a \expectation{X}} \leq \frac{1}{a}$.
\end{proof}

\section{Chebyshev's Inequality and the Chernoff Bound}

Chebyshev's inequality uses the variance of the random variable, which is also called the
\emph{second moment}.

\begin{definition}[Variance and Standard Deviation]
    The \textbf{variance} of a random variable $X$ is
    $\variance{X} = \expectation{\left(X - \expectation{X}\right)^2}$.
    The \textbf{standard deviation} of $X$ is $\sigma = \sqrt{\variance{X}}$.
\end{definition}

\begin{theorem}[Chebyshev's Inequality]\label{thm:chebyshev}
    For any random variable $X$ and any $a > 0$,
    \[\prob{\abs{X - \expectation{X}} \geq a} \leq \frac{\variance{X}}{a^2} = \frac{\sigma^2}{a^2}.\]
\end{theorem}

\begin{proof}
    \[\prob{\abs{X - \expectation{X}} \geq a}
        = \prob{\left(X - \expectation{X}\right)^2 \geq a^2}
        \leq \frac{\expectation{\left(X - \expectation{X}\right)^2}}{a^2}
        = \frac{\variance{X}}{a^2},\]
    where the inequality is Markov's inequality applied to the random variable
    $\left(X - \expectation{X}\right)^2 \geq 0$.
\end{proof}

\paragraph{Chernoff bound.}

The Chernoff bounds of the last lecture (\cref{thm:chernoff-0901}) need
$X = \sum_{i=1}^{n} X_i$ to be a sum of \emph{mutually
independent} indicator random variables. Then $\mu = \expectation{X} = \sum_{i=1}^{n} p_i$ with
$p_i = \prob{X_i = 1}$ by linearity of expectation.

\section{Search}

Let $f \colon D \to R$ be a function from a domain $D$ to a range $R$.
Given $y$, we ask: is there an $x$ such that $f(x) = y$? We consider
$f \colon [1, \dots, N] \to [1, \dots, N]$ and want to find an $x$ such that $f(x) = k$. As in the
earlier lectures, the function can be seen as an array with $n$ slots (\cref{fig:coinflips-0825}),
where $n$ is the size of the domain, and the cost of an algorithm is the number of queries to $f$. A
randomized classical algorithm needs $\bigTh{n}$ queries, and Grover's algorithm needs
$\bigO{\sqrt{n}}$ queries. For the random sampling algorithm,
$\expectation{\text{queries}} = \bigO{n}$ and one sample has $\prob{\text{Success}} = \sfrac{1}{n}$.

\paragraph{Probability amplitude amplification.} If the classical success probability is $p$, then
in the quantum case it becomes $\sqrt{p}$.

\section{Birthday Paradox}

There are $m$ people and $n$ birthdays ($n = 365$), and $f(x)$ is the birthday of person $x$. We
are looking for a \emph{collision}, i.e., $x \neq y$ such that $f(x) = f(y)$.

Let $E_i$ be the event that the $\nth{i}$ birthday is different from the first $i-1$ birthdays.
By the chain rule and $1 - x \leq e^{-x}$,
\begin{align*}
    \prob{\text{all birthdays different}}
     & = \prob{E_1} \cdot \prob{E_2 \given E_1} \cdot \prob{E_3 \given E_2 \cap E_1} \cdots
    = \prod_{i=0}^{m-1} \left(1 - \frac{i}{n}\right)                                        \\
     & \leq \prod_{i=0}^{m-1} e^{-\sfrac{i}{n}}
    = e^{-\frac{(m-1)m}{2n}}.
\end{align*}
For $m = \sqrt{2n} + 1$ we have $(m-1)m \geq 2n$, so this is at most $\sfrac{1}{e} \leq \sfrac{1}{2}$.
Hence $\bigO{\sqrt{n}}$ people are enough to have a constant probability of collision.

\subsection{Expected Number of Collisions}

The birthday paradox can be seen as throwing $m$ balls into $n$ bins.
We show that for $m \approx \sqrt{2n}$,
$\expectation{\text{\# collisions}} \geq 1$.

Let $X$ be the number of collisions. $X$ takes values from $0$ to $\binom{m}{2}$. Write
\[X = \sum_{i \neq j} X_{ij},
    \qquad
    X_{ij} = \begin{cases}
        1 & \text{if } f(i) = f(j), \\
        0 & \text{else,}
    \end{cases}\]
where the sum is over the $\binom{m}{2}$ pairs $i \neq j$. The random variables $X_{ij}$ are
\emph{pairwise independent}. By linearity of expectation,
\[\expectation{X} = \sum_{i \neq j} \prob{X_{ij} = 1} = \binom{m}{2} \frac{1}{n}
    = \frac{m(m-1)}{2n} \sim \frac{m^2}{2n} = \bigTh{\frac{m^2}{n}}.\]

\begin{exercise}\label{ex:birthday-0903}
    Show that
    \[\prob{X \geq 1} \geq \bigTh{1},\]
    i.e., the birthday paradox occurs.
\end{exercise}

We compute $\variance{X}$ and use Chebyshev's inequality (\cref{thm:chebyshev}).

\begin{lemma}[Variance of the Number of Collisions]\label{lem:pairwise-0903}
    For any two different pairs $\set{i,j} \neq \set{k,l}$,
    \[\expectation{X_{ij} X_{kl}} = \expectation{X_{ij}} \expectation{X_{kl}} = \frac{1}{n^2},\]
    i.e., $X_{ij}$ and $X_{kl}$ are independent. Consequently
    \[\variance{X} = \sum_{i \neq j} \variance{X_{ij}}
        = \binom{m}{2} \frac{1}{n} \left(1 - \frac{1}{n}\right) \leq \expectation{X}.\]
\end{lemma}

\begin{proof}
    We have $\expectation{X_{ij} X_{kl}} = \prob{f(i) = f(j) \text{ and } f(k) = f(l)}$. If the
    pairs are disjoint, the two events depend on the birthdays of four different people, which are
    independent, so the probability is $\prob{f(i) = f(j)} \cdot \prob{f(k) = f(l)} = \sfrac{1}{n^2}$.
    Otherwise the pairs share one person, say $\set{i,j}$ and $\set{i,k}$ with $j \neq k$. Then
    $f(j)$ and $f(k)$ are independent of $f(i)$ and of each other, and each equals a given day $b$
    with probability $\sfrac{1}{n}$, so
    \[\prob{f(j) = f(i) \text{ and } f(k) = f(i)}
        = \sum_{b=1}^{n} \prob{f(i) = b} \cdot \frac{1}{n} \cdot \frac{1}{n}
        = \frac{1}{n^2}.\]
    In both cases $\expectation{X_{ij} X_{kl}} = \sfrac{1}{n^2} = \expectation{X_{ij}} \expectation{X_{kl}}$.
    For two indicators this is the same as independence, since their joint distribution is
    determined by $\prob{X_{ij} = 1}$, $\prob{X_{kl} = 1}$ and $\prob{X_{ij} = X_{kl} = 1}$.

    For the variance, expand the square of
    $X - \expectation{X} = \sum_{i \neq j} \left(X_{ij} - \expectation{X_{ij}}\right)$:
    \begin{align*}
        \variance{X}
         & = \sum_{i \neq j} \expectation{\left(X_{ij} - \expectation{X_{ij}}\right)^2}
        + \sum_{\set{i,j} \neq \set{k,l}} \expectation{\left(X_{ij} - \expectation{X_{ij}}\right) \left(X_{kl} - \expectation{X_{kl}}\right)} \\
         & = \sum_{i \neq j} \variance{X_{ij}}
        + \sum_{\set{i,j} \neq \set{k,l}} \left(\expectation{X_{ij} X_{kl}} - \expectation{X_{ij}} \expectation{X_{kl}}\right)
        = \sum_{i \neq j} \variance{X_{ij}},
    \end{align*}
    since the second sum vanishes by the first part. As $X_{ij}^2 = X_{ij}$,
    $\variance{X_{ij}} = \expectation{X_{ij}^2} - \expectation{X_{ij}}^2
        = \frac{1}{n} \left(1 - \frac{1}{n}\right) \leq \frac{1}{n}$,
    and summing over the $\binom{m}{2}$ pairs gives
    $\variance{X} \leq \binom{m}{2} \frac{1}{n} = \expectation{X}$.
\end{proof}

\begin{theorem}[The Birthday Paradox Occurs]\label{thm:birthday-0903}
    If $m \geq 2\sqrt{n} + 1$, then
    \[\prob{X \geq 1} \geq \frac{1}{2}.\]
    In words: with $\bigO{\sqrt{n}}$ people there is a collision with constant probability.
\end{theorem}

\begin{proof}
    If $X = 0$ then $\abs{X - \expectation{X}} = \expectation{X}$. So by Chebyshev's inequality
    (\cref{thm:chebyshev}) with $a = \expectation{X}$, and then \cref{lem:pairwise-0903},
    \[\prob{X = 0} \leq \prob{\abs{X - \expectation{X}} \geq \expectation{X}}
        \leq \frac{\variance{X}}{\expectation{X}^2} \leq \frac{1}{\expectation{X}}.\]
    For $m \geq 2\sqrt{n} + 1$, since $m(m-1)$ grows with $m$,
    $\expectation{X} = \frac{m(m-1)}{2n} \geq \frac{(2\sqrt{n} + 1) \cdot 2\sqrt{n}}{2n} \geq 2$,
    so $\prob{X = 0} \leq \sfrac{1}{2}$ and $\prob{X \geq 1} = 1 - \prob{X = 0} \geq \sfrac{1}{2}$.
\end{proof}

\section{Quantum Birthday Paradox}

The birthday paradox solves the \emph{collision problem}: given query access
to a function $f$, find $x \neq y$ with $f(x) = f(y)$. Querying $m = \sqrt{2n} + 1$ random points finds
a collision with probability at least $\sfrac{1}{2}$, so $\bigO{\sqrt{n}}$ queries suffice classically.
The quantum version is studied later in the course. Here we show that classically
$\sqrt{n}$ queries are also necessary. An algorithm makes $k$ queries $f(x_1), \dots, f(x_k)$,
where the point $x_s$ may depend on the earlier answers $f(x_1), \dots, f(x_{s-1})$ and on random coins.

\begin{exercise}\label{ex:collision-lb-0903}
    Show a $\sqrt{n}$ lower bound in the classical case.
\end{exercise}

As in the birthday paradox, $f \colon D \to [n]$, where $[n] = \set{1, \dots, n}$, is a
\emph{random function}, i.e., every value $f(x)$ is uniform in $[n]$ and independent of all the
other values. The algorithm \emph{finds a collision} if it outputs $x \neq y$ with $f(x) = f(y)$.
We may assume that it queries the two points that it outputs (at most $2$ extra queries, which
does not change the bound below) and that the $k$ queried points are different, since repeating a
query gives no new information.

\begin{theorem}[Classical Lower Bound for Collision Finding]\label{thm:collision-lb-0903}
    Let $f \colon D \to [n]$ be a random function. Every classical algorithm (deterministic or
    randomized, adaptive) that makes $k$ queries finds a collision with probability at most
    \[\frac{\binom{k}{2}}{n} = \frac{k(k-1)}{2n}.\]
    In particular, to find a collision with probability at least $\sfrac{1}{2}$ it needs
    $k > \sqrt{n}$ queries, so the classical query complexity of the collision problem is $\bigOm{\sqrt{n}}$.
\end{theorem}

\begin{proof}
    Since the algorithm queries the points that it outputs, it can only find a collision if two of
    its queried values agree, i.e., if for some $s$ the event $E_s$ that
    $f(x_s) \in \set{f(x_1), \dots, f(x_{s-1})}$ occurs. Condition on the first $s-1$ answers and on
    the random coins of the algorithm. Then $x_s$ is a fixed point that has not been queried before,
    so $f(x_s)$ is still uniform in $[n]$ and independent of the $s-1$ values
    $f(x_1), \dots, f(x_{s-1})$, which are now fixed. Hence it equals one of them with probability
    at most $\frac{s-1}{n}$, no matter how the algorithm chose $x_s$. As this holds for every
    outcome of the first $s-1$ steps, $\prob{E_s} \leq \frac{s-1}{n}$ without conditioning, and by
    the union bound
    \[\prob{\text{collision}} \leq \sum_{s=1}^{k} \prob{E_s} \leq \sum_{s=1}^{k} \frac{s-1}{n} = \frac{k(k-1)}{2n}.\]
    If the algorithm succeeds with probability at least $\sfrac{1}{2}$, then $\frac{k(k-1)}{2n} \geq \frac{1}{2}$,
    i.e., $k^2 > k(k-1) \geq n$, so $k > \sqrt{n}$.
\end{proof}

\paragraph{Worst case inputs.} \cref{thm:collision-lb-0903} is about a \emph{random} input $f$.
By Yao's minmax principle from the lecture of September~1 (in its version for algorithms that are
allowed to err), the bound $\bigOm{\sqrt{n}}$ also holds for every randomized algorithm on its
worst case input. With the birthday paradox, the classical query complexity of finding a collision
is $\bigTh{\sqrt{n}}$.
